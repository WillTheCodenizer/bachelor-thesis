\section{Introduction}

\subsection{Fast Radio Bursts}
\begin{frame}{Fast radio bursts: bright millisecond flashes from far away}
  \begin{columns}[c]
    \begin{column}{0.44\textwidth}
      \begin{itemize}
        \item Radio pulses lasting $\lesssim\,$ms
        \item Peak flux densities from \qty{50}{\milli\jansky} to \qty{100}{\jansky}
        \item Detected from \qty{110}{\mega\hertz} to \qty{8}{\giga\hertz}
        \item One burst releases as much energy as the Sun in \alert{three days}
        \item Received signal $\approx 10^{3}\times$ weaker than a \alert{mobile phone on the Moon}
        \item First detection: Lorimer Burst (2007)
        \item Second CHIME/FRB catalog: \num{4539} bursts from \num{3641} sources, 83 repeaters
      \end{itemize}
    \end{column}
    \begin{column}{0.54\textwidth}
      \centering
      \includegraphics[width=\linewidth]{frb/frb_sky_distribution_petroff.png}
      \plotlabel{Known FRBs in Galactic coordinates (Aitoff projection), up to 2022}
      \source{Adapted from Petroff et al.\ 2022; Petroff et al.\ 2019; Lorimer et al.\ 2007; CHIME/FRB Collaboration 2026}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{The dispersion measure reveals an extragalactic origin}
  Lower frequencies arrive later; the delay defines the dispersion measure (DM)
  \begin{equation*}
    \mathrm{DM} = \frac{2\pi m_{\mathrm{e}} c}{e^2}\, \Delta t
    \left( \frac{1}{\nu_{\mathrm{low}}^2} - \frac{1}{\nu_{\mathrm{high}}^2} \right)^{-1}
    \qquad
    \mathrm{DM} \propto \int_0^{L} \mathrm{d}s\, n_{\mathrm{e}}(s)
  \end{equation*}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Milky Way}
        Galactic electron-density models: \\
        $\mathrm{DM} \lesssim \qty{50}{\parsec\per\cubic\centi\metre}$
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Observed FRBs}
        Often \\
        $\mathrm{DM} \gtrsim \qty{1000}{\parsec\per\cubic\centi\metre}$
      \end{block}
    \end{column}
  \end{columns}
  \bigskip
  \begin{itemize}
    \item Excess DM $\Rightarrow$ sources at \alert{cosmological distances}
    \item Approximately isotropic on the sky $\Rightarrow$ extragalactic population
    \item FRBs therefore trace the \alert{large-scale structure} of the Universe
  \end{itemize}
  \source{Petroff et al.\ 2019; Reischke et al.\ 2021}
\end{frame}

\subsection{Progenitor Mechanisms}
\begin{frame}{Two competing progenitor channels}
  Repeating sources rule out a purely cataclysmic origin for at least part of the population.
  \medskip
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Young magnetars \hfill (prompt)}
        \small
        \begin{itemize}
          \item Neutron stars with extreme magnetic fields
          \item Born in core-collapse supernovae, \qtyrange{3}{48}{\Myr} after star formation
          \item Formation rate follows the cosmic star-formation history
          \item FRB-like burst from Galactic magnetar SGR~1935+2154 (2020)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Neutron star mergers \hfill (delayed)}
        \small
        \begin{itemize}
          \item FRB~20180924A, FRB~20190523A: massive, low-SFR hosts, large offsets
          \item FRB~20200120E in a globular cluster
          \item Delay times up to several Gyr
          \item Population evolves slowly with redshift
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \bigskip
  \centering
  This work: the difference is encoded in a \alert{redshift-dependent bias $b(z)$}
  \source{Petroff et al.\ 2019; CHIME/FRB Collaboration 2020; Wang et al.\ 2020; Zapartas et al.\ 2017; Ruiter et al.\ 2019; Lin et al.\ 2022}
\end{frame}

\subsection{Research Question}
\begin{frame}{Idea: let the clustering tell us where FRBs live}
  \begin{itemize}
    \item Host identification of single FRBs needs arcsecond localizations $\Rightarrow$ rare
    \item FRBs trace the large-scale structure $\Rightarrow$ their \alert{angular clustering} encodes the
      bias of the host population
    \item Host bias depends on the progenitor channel
    \item Problem: FRB samples are sparse $\Rightarrow$ strong \alert{shot noise}
    \item Solution: cross-correlate with a \alert{dense photometric galaxy sample} (KiDS-Legacy)
  \end{itemize}
  \bigskip
  \begin{block}{Research question}
    How well can the FRB host bias be constrained by cross-correlating FRBs with KiDS-Legacy,
    and can a prompt magnetar origin be distinguished from a delayed neutron-star origin?
  \end{block}
  \medskip
  Shallow, CHIME-like and deep, SKA-like FRB survey $\times$ two progenitor models $\to$ Fisher forecast
  \source{Rafiei-Ravandi et al.\ 2021; Reischke et al.\ 2021}
\end{frame}

\subsection{Cosmological Background}
\begin{frame}{A flat $\symup{\Lambda}$CDM universe sets the distance scale}
  \begin{equation*}
    \mathrm{d}s^2 = -c^2 \mathrm{d}t^2 + a^2(t) \left[ \frac{\mathrm{d}r^2}{1 - k r^2}
    + r^2 \left(\mathrm{d}\theta^2 + \sin^2\theta\, \mathrm{d}\phi^2\right) \right]
    \qquad
    1+z = \frac{a(t_0)}{a(t_{\mathrm{e}})}
  \end{equation*}
  \begin{columns}[c]
    \begin{column}{0.48\textwidth}
      Late-time expansion rate (flat $\symup{\Lambda}$CDM)
      \begin{equation*}
        E(z) \approx \sqrt{\symup{\Omega}_{\mathrm{m},0}(1+z)^3 + \symup{\Omega}_{\symup{\Lambda},0}}
      \end{equation*}
    \end{column}
    \begin{column}{0.48\textwidth}
      Line-of-sight comoving distance
      \begin{equation*}
        \chi(z) = \chi_{\mathrm{H}} \int_0^z \frac{\mathrm{d}z'}{E(z')}
      \end{equation*}
    \end{column}
  \end{columns}
  \bigskip
  \begin{itemize}
    \item $\chi(z)$ maps redshift distributions onto the line of sight
    \item $H(z)/c = \mathrm{d}z/\mathrm{d}\chi$ converts kernels from redshift to comoving distance
  \end{itemize}
  \source{Hogg 2000; Planck Collaboration 2018}
\end{frame}

\subsection{Structure Formation}
\begin{frame}{Tracers are biased versions of the matter field}
  \begin{columns}[c]
    \begin{column}{0.5\textwidth}
      Density contrast and linear growth
      \begin{equation*}
        \delta(\mathbf{x}) := \frac{\rho(\mathbf{x}) - \bar{\rho}}{\bar{\rho}}
        \qquad
        \delta(\mathbf{x}, z) = D_+(z)\, \delta(\mathbf{x}, 0)
      \end{equation*}
      Power spectrum
      \begin{equation*}
        \left\langle \hat{\delta}(\mathbf{k})\, \hat{\delta}^{*}(\mathbf{k}') \right\rangle
        = (2\pi)^3 \delta_{\mathrm{D}}(\mathbf{k}-\mathbf{k}')\, P(k)
      \end{equation*}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{Linear bias}
        \begin{equation*}
          \delta_{\mathrm{t}}(\mathbf{x}, z) = b(z)\, \delta(\mathbf{x}, z)
        \end{equation*}
      \end{block}
      \begin{itemize}
        \item Tracers form in density peaks
        \item $b > 1$: more strongly clustered than matter
        \item $b(z)$ tells us \alert{which halos host the FRBs}
      \end{itemize}
    \end{column}
  \end{columns}
  \source{Bernardeau et al.\ 2002; Desjacques et al.\ 2018}
\end{frame}

\subsection{Angular Power Spectrum}
\begin{frame}{Projecting onto the sky: the angular power spectrum}
  Limber approximation for two tracers $A$, $B$ (valid for $\ell \gtrsim 10$)
  \begin{equation*}
    C_{\ell}^{AB} \approx \int \frac{\mathrm{d}\chi}{\chi^2}\, W^{A}(\chi)\, W^{B}(\chi)\,
    P_{AB}\!\left(k = \frac{\ell+1/2}{\chi},\, z(\chi)\right)
    \qquad
    P_{AB}(k, z) = b_A(z)\, b_B(z)\, P(k, z)
  \end{equation*}
  Radial weight functions
  \begin{equation*}
    W_{\mathrm{FRB}}(z) = b(z)\, n(z)
    \qquad
    W_{\mathrm{gal}}^{\,i}(z) = b_{\mathrm{g}}^{\,i}\, n_i(z)
    \qquad
    W(\chi) = W(z)\, \frac{H(z)}{c}
  \end{equation*}
  \begin{block}{Key point}
    The FRB$\times$galaxy cross-spectrum measures $b_{\mathrm{FRB}}\, b_{\mathrm{g}}^{\,i}$ where the two
    radial kernels overlap $\Rightarrow$ each tomographic bin probes the FRB bias at a different redshift.
  \end{block}
  \source{Limber 1953; LoVerde \& Afshordi 2008; Khosravi et al.\ 2016}
\end{frame}

\subsection{Fisher Matrix}
\begin{frame}{Fisher forecast for the FRB host bias}
  \begin{equation*}
    F_{ij} = f_{\mathrm{sky}} \sum_{\ell=\ell_{\mathrm{min}}}^{\ell_{\mathrm{max}}} \frac{2\ell+1}{2}\,
    \operatorname{Tr}\left[ C_{\ell}^{-1}\, \frac{\partial C_{\ell}}{\partial \theta_i}\,
    C_{\ell}^{-1}\, \frac{\partial C_{\ell}}{\partial \theta_j} \right]
  \end{equation*}
  \begin{columns}[t]
    \begin{column}{0.5\textwidth}
      \vspace{-2ex}
      \begin{equation*}
        \frac{\partial C_{\ell}}{\partial \theta_i} = \frac{C_{\ell}(\theta_i + h_i) - C_{\ell}(\theta_i - h_i)}{2 h_i}
      \end{equation*}
      \begin{itemize}
        \item $C_\ell$: matrix of all auto- and cross-spectra, shot noise in the auto-spectra
        \item $\theta = (b_0, \delta)$, step $h_i = 0.01\,\theta_i$
        \item Cosmology fixed to Planck 2018
      \end{itemize}
    \end{column}
    \begin{column}{0.46\textwidth}
      \vspace{-2ex}
      \begin{equation*}
        \mathrm{Cov} = F^{-1}
        \qquad
        \sigma_i = \sqrt{\mathrm{Cov}_{ii}}
      \end{equation*}
      \begin{equation*}
        \mathrm{FoM} = \frac{1}{\sqrt{\det \mathrm{Cov}}}
      \end{equation*}
    \end{column}
  \end{columns}
  \medskip
  \begin{block}{Sky fractions}
    FRB-only: $f_{\mathrm{sky}} = 0.9$ \qquad
    FRB$\times$galaxy: $f_{\mathrm{sky}} = 0.0327$ (KiDS) $\Rightarrow$ \alert{$\approx 28\times$ smaller area}
  \end{block}
  \source{Tegmark et al.\ 1997; Reischke et al.\ 2021; Coe 2009; Wang 2008}
\end{frame}

\subsection{Limitations}
\begin{frame}{Limitations}
  \begin{itemize}
    \item Limber approximation restricts the analysis to $\ell \geq 10$, excluding the largest scales
    \item Gaussian likelihood and covariance neglect non-Gaussian contributions \\
      $\Rightarrow$ uncertainties are \alert{optimistic lower bounds}
    \item Power-law bias adequate for $z \lesssim 1.5$, where the constraints come from;
      at higher $z$ the bias evolves more strongly and depends on halo mass
    \item FRB redshift distribution and fiducial bias values are model assumptions, not measurements
    \item Survey geometry only through $f_{\mathrm{sky}}$; a real analysis needs pseudo-spectra
  \end{itemize}
  \source{Berti et al.\ 2021; Pan et al.\ 2020; Tinker et al.\ 2010; Hivon et al.\ 2002}
\end{frame}

\subsection{Summary}
\begin{frame}{Conclusion}
  \begin{enumerate}
    \item Forecasting pipeline: nonlinear $P(k,z)$ $\to$ Limber auto- and cross-spectra of FRBs and
      six KiDS-Legacy bins $\to$ Fisher forecast for $b_0$ and $\delta$
    \item FRB auto-correlation alone is \alert{shot-noise dominated} and does not constrain the bias
    \item Cross-correlation with KiDS: $\sigma$ reduced \alert{15--46$\times$}, FoM increased
      \alert{51--377$\times$}, despite a $\approx 28\times$ smaller sky area
    \item KiDS footprint not yet sufficient to separate prompt magnetars from delayed neutron stars
    \item LSST Y10: FoM $\times$ 9--12, separation becomes comparable to the uncertainties
  \end{enumerate}
  \bigskip
  \begin{block}{Take-home message}
    FRB$\times$galaxy cross-correlation is a promising probe of FRB progenitors, and next-generation
    galaxy surveys can make it decisive.
  \end{block}
\end{frame}

\begin{frame}{Backup: full-sky spectrum and the Limber approximation}
  Exact full-sky angular cross-power spectrum
  \begin{equation*}
    C_{\ell}^{AB} = \frac{2}{\pi} \int_0^\infty k^2\,\mathrm{d}k
    \int \mathrm{d}\chi\, W^{A}(\chi)\, j_{\ell}(k\chi)
    \int \mathrm{d}\chi'\, W^{B}(\chi')\, j_{\ell}(k\chi')\, P_{AB}(k;\chi,\chi')
  \end{equation*}
  \begin{itemize}
    \item Spherical Bessel functions $j_\ell$ peak sharply at $k\chi \approx \ell + 1/2$ for large $\ell$
    \item Kernels vary slowly compared to $j_\ell$ $\Rightarrow$ only $k = (\ell+1/2)/\chi$ contributes
  \end{itemize}
  \begin{equation*}
    \Rightarrow\quad
    C_{\ell}^{AB} \approx \int \frac{\mathrm{d}\chi}{\chi^2}\, W^{A}(\chi)\, W^{B}(\chi)\,
    P_{AB}\!\left(\frac{\ell+1/2}{\chi},\, z(\chi)\right)
  \end{equation*}
  \begin{itemize}
    \item Valid for $\ell \gtrsim 10$ $\Rightarrow$ $\ell_{\mathrm{min}} = 10$
    \item $\ell_{\mathrm{max}} = 1000$: nonlinear $P(k)$ prescription still sufficiently reliable
  \end{itemize}
  \source{Limber 1953; LoVerde \& Afshordi 2008}
\end{frame}
